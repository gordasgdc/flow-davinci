cmake_minimum_required(VERSION 3.16)
project(PaneFloatSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(pane-float-smoke main.cpp)
target_link_libraries(pane-float-smoke PRIVATE Qt5::Widgets)
if(WIN32)
    target_link_libraries(pane-float-smoke PRIVATE shell32 ole32 uuid dwmapi comctl32)
endif()
