cmake_minimum_required(VERSION 3.20)
project(Unbound LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Unbound requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(value UNBOUND_VERSION UNBOUND_YEAR UNBOUND_MONTH UNBOUND_DAY UNBOUND_BUILD)
    if(NOT DEFINED ${value})
        message(FATAL_ERROR "Use build-unbound-release.cmd to supply ${value}")
    endif()
endforeach()
find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Widgets)
set(UNBOUND_FILE_TYPE VFT_APP)
set(UNBOUND_FILENAME Unbound.exe)
configure_file(version.rc.in launcher.rc @ONLY)
set(UNBOUND_FILE_TYPE VFT_DLL)
set(UNBOUND_FILENAME Unbound.dll)
configure_file(version.rc.in helper.rc @ONLY)
configure_file(icon.rc.in icon.rc @ONLY)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/icon.rc PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/unbound.ico")
add_executable(Unbound WIN32 launcher.cpp ${CMAKE_CURRENT_BINARY_DIR}/launcher.rc ${CMAKE_CURRENT_BINARY_DIR}/icon.rc)
set_property(TARGET Unbound PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(Unbound PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX UNBOUND_VERSION=L"${UNBOUND_VERSION}")
target_link_libraries(Unbound PRIVATE version)
add_executable(UnboundLauncherTests ../tests/launcher/main.cpp)
set_property(TARGET UnboundLauncherTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(UnboundLauncherTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(UnboundLauncherTests PRIVATE version)
add_library(UnboundHelper SHARED controller.cpp ${CMAKE_CURRENT_BINARY_DIR}/helper.rc)
set_target_properties(UnboundHelper PROPERTIES OUTPUT_NAME Unbound)
set_property(TARGET UnboundHelper PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(UnboundHelper PRIVATE Qt5::Widgets shell32 ole32 uuid dwmapi comctl32)
install(TARGETS Unbound UnboundHelper RUNTIME DESTINATION .)
